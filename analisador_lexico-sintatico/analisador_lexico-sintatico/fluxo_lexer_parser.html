Linguagem analisada Mini-linguagem imperativa, no estilo C

É a linguagem que o lexer e o parser reconhecem: programas formados por atribuições e condicionais if, com expressões aritméticas e relacionais.

Palavras reservadasif int
Valoresinteiros 42 · cadeias "Ana" · identificadores x1
Aritméticos+ - * / =
Relacionais== != < <= > >=
Delimitadores( ) { } ;
x = 2 + 3 * 4;
nome = "Ana Paula";

    y = x - 1;
}

int é reconhecido pelo lexer, mas a gramática não tem regra de declaração. A gramática completa aparece na seção do parser.

03 _peek · _advance · _skip_whitespace lexer.py:183–224

As primitivas. Todo o resto da fase é construído sobre elas — e é aqui que linha e coluna são mantidas em dia. 

def _is_at_end(self) -> bool:
    return self._position >= len(self._source)

def _peek(self) -> str:
    if self._is_at_end():
        return ""
    return self._source[self._position]

def _advance(self) -> str:
    index = self._position
    character = self._source[index]
    self._position += 1

    if character == "\r":
        # Trata CRLF como uma única quebra de linha.
        if self._position < len(self._source) and self._source[self._position] == "\n":
            self._position += 1
        self._line += 1
        self._column = 1
    elif character == "\n":
        self._line += 1
        self._column = 1
    else:
        self._column += 1

    return character

def _skip_whitespace(self) -> None:
    while self._peek() in WHITESPACE_CHARACTERS:
        self._advance()

06 next_token() núcleo lexer.py:149

O despacho: olha um caractere e decide qual scanner chamar. Todo o resto da fase pende desta decisão. 

def next_token(self) -> Token:
    self._skip_whitespace()

    if self._is_at_end():
        return Token(TokenType.EOF, "", None, self._line, self._column)

    character = self._peek()

    if self._is_identifier_start(character):
        return self._scan_identifier()

    if self._is_digit(character):
        return self._scan_number()

    if character == '"':
        return self._scan_string()

    if character in COMPARISON_STARTERS:
        return self._scan_comparison_or_assignment()

    if character in SINGLE_CHAR_TOKENS:
        return self._scan_single_char_token()

    return self._scan_unknown()

O padrão de todos os scanners, na sua forma mais limpa — e o ponto onde palavra reservada e identificador se separam. 

def _scan_identifier(self) -> Token:
    start_position = self._position
    start_line = self._line
    start_column = self._column

    while self._is_identifier_part(self._peek()):
        self._advance()

    lexeme = self._source[start_position:self._position]
    token_type = RESERVED_WORDS.get(lexeme, TokenType.IDENTIFICADOR)
    return Token(token_type, lexeme, lexeme, start_line, start_column)

def _scan_number(self) -> Token:
    start_position = self._position
    start_line = self._line
    start_column = self._column

    while self._is_digit(self._peek()):
        self._advance()

    # Em "12abc", considera-se um único lexema malformado.
    if self._is_identifier_start(self._peek()):
        while self._is_identifier_part(self._peek()):
            self._advance()
        lexeme = self._source[start_position:self._position]
        return Token(TokenType.ERRO, lexeme, None, start_line, start_column)

    lexeme = self._source[start_position:self._position]
    return Token(TokenType.NUM_INTEIRO, lexeme, int(lexeme), start_line, start_column)

A gramática que os blocos 15–18 implementam, uma função por regra: 

<programa>     ::= { <comando> }
<comando>      ::= <atribuicao> | <condicional>
<atribuicao>   ::= <identificador> "=" <expressao> ";"
<condicional>  ::= "if" "(" <expressao_relacional> ")" "{" { <comando> } "}"
<expressao_relacional> ::= <expressao> [ <op_relacional> <expressao> ]
<expressao>    ::= <termo> { ("+" | "-") <termo> }
<termo>        ::= <fator> { ("*" | "/") <fator> }
<fator>        ::= <numero> | <cadeia> | <identificador> | "(" <expressao> ")"

13 _peek · _check · _match · _consume parser.py:418–479

As primitivas, e o vocabulário de todos os blocos seguintes: _check consulta sem consumir, _match consome se casar, _consume exige. 

def _peek(self) -> Token:
    return self.tokens[self.current]

def _previous(self) -> Token:
    return self.tokens[self.current - 1]

def _advance(self) -> Token:
    token = self._peek()
    if token.type is not TokenType.EOF:
        self.current += 1
    return token

def _check(self, token_type: TokenType) -> bool:
    return self._peek().type is token_type

def _match(self, *token_types: TokenType) -> bool:
    if self._peek().type in token_types:
        self._advance()
        return True
    return False

def _consume(self, token_type: TokenType, expected: str) -> Token:
    if self._check(token_type):
        return self._advance()
    self._error(self._peek(), expected)

def _is_reserved(self, word: str) -> bool:
    token = self._peek()
    return token.type is TokenType.PALAVRA_RESERVADA and token.lexeme == word

O laço externo — a regra <programa> — e o ponto onde um erro é capturado para que a análise continue. 

def parse(self) -> tuple[ASTNode, list[SyntaxErrorInfo]]:
    program = ASTNode("PROGRAMA")

    try:
        while not self._check(TokenType.EOF):
            try:
                program.filhos.append(self.command())
            except ParserError as error:
                self._record(error)
                self._synchronize(stop_at_block_end=False)
    except _AbortParsing:
        pass

    return program, self.errors

A tradução linha a linha de uma regra da gramática — e a construção do primeiro nó da AST. 

# <atribuicao> ::= <identificador> "=" <expressao> ";"
def assignment(self) -> ASTNode:
    identifier = self._consume(TokenType.IDENTIFICADOR, "identificador")
    self._consume(TokenType.ATRIBUICAO, "'='")
    expression = self.expression()
    self._consume(TokenType.PONTO_VIRGULA, "';'")

    return ASTNode(
        "ATRIBUICAO",
        filhos=[
            ASTNode("IDENTIFICADOR", identifier.lexeme),
            expression,
        ],
    )
17 conditional() · relational_expression() · _block_commands() parser.py:212 · 242 · 261

A regra composta do if. É em _block_commands() que o parser volta a chamar command() — a porta para o aninhamento. 

# <condicional> ::= "if" "(" <expressao_relacional> ")" "{" { <comando> } "}"
def conditional(self) -> ASTNode:
    with self._nesting(self._peek()):
        self._consume_reserved("if")
        self._consume(TokenType.ABRE_PAREN, "'(' após 'if'")
        condition = self.relational_expression()
        self._consume(TokenType.FECHA_PAREN, "')' após a expressão do if")
        self._consume(TokenType.ABRE_CHAVE, "'{' após ')'")

        body = self._block_commands()
        self._consume(TokenType.FECHA_CHAVE, "'}'")

        return ASTNode("CONDICIONAL", filhos=[condition, ASTNode("BLOCO", filhos=body)])


# <expressao_relacional> ::= <expressao> [ <op_relacional> <expressao> ]
def relational_expression(self) -> ASTNode:
    node = self.expression()
    if self._match(*_RELATIONAL_OPERATORS):
        operator = self._previous().lexeme
        right = self.expression()
        node = ASTNode("RELACIONAL", operator, [node, right])
    return node


def _block_commands(self) -> list[ASTNode]:
    commands: list[ASTNode] = []

    while not self._check(TokenType.FECHA_CHAVE):
        if self._check(TokenType.EOF):
            self._error(self._peek(), "'}' para fechar o bloco do if")

        try:
            commands.append(self.command())
        except ParserError as error:
            self._record(error)
            self._synchronize(stop_at_block_end=True)

    return commands
18 expression() · term() · factor() núcleo parser.py:288 · 312 · 334

A cascata. Três funções em sequência, e é dessa hierarquia de chamadas — não de nenhuma comparação de prioridade — que sai a precedência dos operadores. O _match em factor() fecha o ciclo de volta ao topo. 

# <expressao> ::= <termo> { ("+" | "-") <termo> }
def expression(self) -> ASTNode:
    with self._nesting(self._peek()):
        node = self.term()
        while self._match(TokenType.OP_SOMA, TokenType.OP_SUBTRACAO):
            operator = self._previous().lexeme
            right = self.term()
            node = ASTNode("EXPRESSAO", operator, [node, right])
        return node


# <termo> ::= <fator> { ("*" | "/") <fator> }
def term(self) -> ASTNode:
    node = self.factor()
    while self._match(TokenType.OP_MULTIPLICACAO, TokenType.OP_DIVISAO):
        operator = self._previous().lexeme
        right = self.factor()
        node = ASTNode("TERMO", operator, [node, right])
    return node


# <fator> ::= <numero> | <cadeia> | <identificador> | "(" <expressao> ")"
def factor(self) -> ASTNode:
    token = self._peek()

    if token.type is TokenType.NUM_INTEIRO:
        self._advance()
        return ASTNode("NUMERO", token.value)

    if token.type is TokenType.CADEIA:
        self._advance()
        return ASTNode("CADEIA", token.value)

    if token.type is TokenType.IDENTIFICADOR:
        self._advance()
        return ASTNode("IDENTIFICADOR", token.lexeme)

    if self._match(TokenType.ABRE_PAREN):
        expression = self.expression()
        self._consume(TokenType.FECHA_PAREN, "')'")
        return ASTNode("GRUPO", filhos=[expression])

    self._error(token, "fator (numero, cadeia, identificador ou expressão entre parênteses)")
19 _error() parser.py:508

A via única de relatório. Lançada de qualquer profundidade da recursão, ela também decide se a mensagem é léxica ou sintática. 

@staticmethod
def _error(token: Token, expected: str) -> NoReturn:
    found = token.lexeme if token.lexeme else "EOF"

    if token.type is TokenType.ERRO:
        message = f"Erro lexico: token invalido '{found}'."
    else:
        message = f"Erro sintatico: esperado {expected}, encontrado '{found}'."

    raise ParserError(SyntaxErrorInfo(message, token.line, token.column, found))
20 _synchronize() parser.py:397

A recuperação. Descarta tokens até um ponto de retomada, e é o que permite reportar vários erros em uma única execução. 

def _synchronize(self, stop_at_block_end: bool) -> None:
    self._advance()

    while not self._check(TokenType.EOF):
        if self._previous().type is TokenType.PONTO_VIRGULA:
            return
        if stop_at_block_end and self._check(TokenType.FECHA_CHAVE):
            return
        if self._is_reserved("if"):
            return
        self._advance()
